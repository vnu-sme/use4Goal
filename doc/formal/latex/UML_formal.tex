\section{Cú pháp hình thức của Class Diagram}

Giả sử
\[
N\subseteq A^{+},
\qquad
\Sigma=(T,\Omega).
\]

Một Class Diagram được xác định bởi
\[
\mathcal{CD}
=
\bigl(
Class,\{Att_c\}_{c\in Class},\{Op_c\}_{c\in Class},
Assoc,associates,roles,multiplicities,\prec
\bigr).
\]


% ============================================================
\subsection{Class}
% ============================================================

\[
Class \subseteq N,
\qquad
\forall c \in Class,\ {include} \, t_c \in T.
\]


% ============================================================
\subsection{Attribute}
% ============================================================

Với mỗi $c\in Class$:
\[
Att_c
=
\{\,a:t_c\to t\mid a\in N,\ t\in T\,\}.
\]

\[
\forall t,t'\in T:
\quad
(a:t_c\to t\in Att_c
\land
a:t_c\to t'\in Att_c)
\Rightarrow t=t'.
\]

Tương tự, tập operation của $c$:
\[
Op_c
=
\left\{
\omega:t_c\times t_1\times\cdots\times t_n\to t
\mid
\omega\in N,\ t,t_1,\ldots,t_n\in T
\right\}.
\]


% ============================================================
\subsection{Association}
% ============================================================

\subsubsection{Association}

\[
Assoc\subseteq N,
\qquad |Assoc|<\infty.
\]

\[
associates:
\left\{
\begin{array}{rcl}
Assoc & \to & Class^{+}\\
as & \mapsto & \langle c_1,\ldots,c_n\rangle,\quad n\geq2
\end{array}
\right.
\]


% ------------------------------------------------------------
\subsubsection{Role name}
% ------------------------------------------------------------

\[
roles:
\left\{
\begin{array}{rcl}
Assoc & \to & N^{+}\\
as & \mapsto & \langle r_1,\ldots,r_n\rangle
\end{array}
\right.
\]

với
\[
associates(as)=\langle c_1,\ldots,c_n\rangle,
\]
và
\[
\forall i,j\in\{1,\ldots,n\}:
\quad
i\neq j\Rightarrow r_i\neq r_j.
\]


% ------------------------------------------------------------
\subsubsection{Participating association}
% ------------------------------------------------------------

\[
participating:
\left\{
\begin{array}{rcl}
Class & \to & \mathcal{P}(Assoc)\\
c & \mapsto &
\{\,as\mid as\in Assoc
\land associates(as)=\langle c_1,\ldots,c_n\rangle
\\
&&
\qquad\land\exists i\in\{1,\ldots,n\}:c_i=c\,\}
\end{array}
\right.
\]


% ------------------------------------------------------------
\subsubsection{Navigable ends}
% ------------------------------------------------------------

\[
navends:
\left\{
\begin{array}{rcl}
Class\times Assoc & \to & \mathcal{P}(N)\\
(c,as) & \mapsto &
\{\,r\mid associates(as)=\langle c_1,\ldots,c_n\rangle
\\
&&
\qquad\land roles(as)=\langle r_1,\ldots,r_n\rangle
\\
&&
\qquad\land\exists i,j\in\{1,\ldots,n\}:
(i\neq j\land c_i=c\land r_j=r)\,\}
\end{array}
\right.
\]

\[
navends:
\left\{
\begin{array}{rcl}
Class & \to & \mathcal{P}(N)\\
c & \mapsto &
\displaystyle\bigcup_{as\in participating(c)}navends(c,as)
\end{array}
\right.
\]


% ------------------------------------------------------------
\subsubsection{Multiplicity}
% ------------------------------------------------------------

\[
multiplicities:
\left\{
\begin{array}{rcl}
Assoc
& \to &
\bigl(\mathcal{P}(\mathbb{N}_0)\setminus\{\varnothing\}\bigr)^{+}
\\
as
& \mapsto &
\langle M_1,\ldots,M_n\rangle
\end{array}
\right.
\]

với
\[
\varnothing\neq M_i\subseteq\mathbb{N}_0,
\qquad
M_i\neq\{0\},
\qquad
1\leq i\leq n.
\]


% ============================================================
\subsection{Kế thừa}
% ============================================================

\[
\prec\ \subseteq Class\times Class
\]
là một thứ tự bộ phận trên $Class$.

\[
c_1\prec c_2
\]
biểu diễn $c_1$ là lớp con của $c_2$.


% ============================================================
\subsection{Các tính chất của kế thừa}
% ============================================================

\[
parents:
\left\{
\begin{array}{rcl}
Class & \to & \mathcal{P}(Class)\\
c & \mapsto &
\{\,c'\mid c'\in Class\land c\prec c'\,\}
\end{array}
\right.
\]

\[
Att_c^{*}
=
Att_c
\cup
\bigcup_{c'\in parents(c)}Att_{c'}.
\]

\[
Op_c^{*}
=
Op_c
\cup
\bigcup_{c'\in parents(c)}Op_{c'}.
\]

\[
navends^{*}:
\left\{
\begin{array}{rcl}
Class & \to & \mathcal{P}(N)\\
c & \mapsto &
navends(c)
\cup
\displaystyle\bigcup_{c'\in parents(c)}navends(c')
\end{array}
\right.
\]

\[
FD_c
=
\bigl(
Att_c^{*},
Op_c^{*},
navends^{*}(c)
\bigr).
\]


% ============================================================
\subsection{Well-Formedness}
% ============================================================

\paragraph{WF-1.}
\[
\forall c\in Class,\;
\forall as_1,as_2\in participating(c):
\]
\[
as_1\neq as_2
\Rightarrow
navends(c,as_1)\cap navends(c,as_2)=\varnothing.
\tag{WF-1}
\]


\paragraph{WF-2.}
\[
\forall c\in Class,\;
\forall
(a:t_{c_1}\to t_1),
(a':t_{c_2}\to t_2)
\in Att_c^{*}:
\]
\[
a=a'
\Rightarrow
(c_1=c_2\land t_1=t_2).
\tag{WF-2}
\]


\paragraph{WF-3.}
\[
\begin{aligned}
&
\omega:t_{c_1}\times t_1\times\cdots\times t_n\to t
\in Op_c^{*}
\\
{}\land{}&
\omega:t_{c_2}\times t_1\times\cdots\times t_n\to t
\in Op_c^{*}
\\
&\Rightarrow c_1=c_2.
\end{aligned}
\tag{WF-3}
\]


\paragraph{WF-4.}
\[
\forall c\in Class,\;
\forall c_1,c_2\in parents(c)\cup\{c\}:
\]
\[
c_1\neq c_2
\Rightarrow
navends(c_1)\cap navends(c_2)=\varnothing.
\tag{WF-4}
\]


\paragraph{WF-5.}
\[
\forall c\in Class,\;
\forall(a:t_{c'}\to t)\in Att_c^{*},\;
\forall r\in navends^{*}(c):
\]
\[
a\neq r.
\tag{WF-5}
\]



% ============================================================
\section{Interpretation of Object Models}
% ============================================================

\subsection{Objects}

Với mỗi $c\in Class$:
\[
oid(c)=\{c_1,c_2,\ldots\}.
\]

Miền ngữ nghĩa của lớp $c$ được xác định bởi
\[
I_{Class}(c)
=
\bigcup
\left\{
oid(c')
\;\middle|\;
c'\in Class \land c'\preceq c
\right\}.
\]

% ============================================================
\subsection{Interpretation of Generalization}
% ============================================================

\[
\forall c_1,c_2\in Class:
\qquad
c_1\prec c_2
\Longrightarrow
I_{Class}(c_1)\subseteq I_{Class}(c_2).
\]


% ============================================================
\subsection{Links}
% ============================================================

Với
\[
associates(as)=\langle c_1,\ldots,c_n\rangle,
\]
miền ngữ nghĩa của association $as$ là
\[
I_{Assoc}(as)
=
I_{Class}(c_1)\times\cdots\times I_{Class}(c_n),
\]

Một link của $as$ là
\[
l_{as}\in I_{Assoc}(as).
\]


% ============================================================

% ============================================================
\subsection{System State}
% ============================================================

\begin{dinhnghia}[System State]
Một trạng thái hệ thống của model $M$ là cấu trúc
\[
\sigma(M)
=
\bigl(
\sigma_{Class},
\sigma_{Att},
\sigma_{Assoc}
\bigr).
\]

\begin{enumerate}
    \item Với mỗi $c\in Class$:
    \[
    \sigma_{Class}(c)\subset oid(c),
    \qquad
    |\sigma_{Class}(c)|<\infty.
    \]

    \item Với mỗi
    \[
    a:t_c\to t\in Att_c^{*},
    \]
    ta có
    \[
    \sigma_{Att}(a):
    \sigma_{Class}(c)\to I(t).
    \]

    \item Với mỗi $as\in Assoc$:
    \[
    \sigma_{Assoc}(as)\subset I_{Assoc}(as),
    \qquad
    |\sigma_{Assoc}(as)|<\infty.
    \]

    Nếu
    \[
    associates(as)=\langle c_1,\ldots,c_n\rangle,
    \]
    thì
    \[
    \forall i\in\{1,\ldots,n\},
    \forall l\in\sigma_{Assoc}(as):
    \]
    \[
    \left|
    \left\{
    l'
    \;\middle|\;
    l'\in\sigma_{Assoc}(as)
    \land
    \bar{\pi}_i(l')=\bar{\pi}_i(l)
    \right\}
    \right|
    \in
    \pi_i\bigl(multiplicities(as)\bigr).
    \]
\end{enumerate}
\end{dinhnghia}

Trong đó $\pi_i(l)$ là phép chiếu thành phần thứ $i$ của $l$ và
$\bar{\pi}_i(l)$ là phép chiếu tất cả các thành phần ngoại trừ thành phần thứ $i$.


\section{Mở rộng}

\[
Class = Entity \cup Role \cup Group \subseteq N,
\]
\[
Entity\cap Role
=
Entity\cap Group
=
Role\cap Group
=
\varnothing,
\qquad
\forall c\in Class,\ \exists!\,t_c\in T.
\]

\[
OrgClass = Role\cup Group.
\]

\[
Composition\subseteq Assoc.
\]

% ============================================================
% Association giữa các intentional classes
% ============================================================

\begin{ghichu}
Phần mở rộng này không đưa vào một quan hệ Owner độc lập, vì khi dịch ACL
sang USE/OCL, Owner luôn trở thành Composition. Do đó, quan hệ sở hữu giữa
một Group và một Role hoặc Group khác được biểu diễn trực tiếp bằng
$Composition\subseteq Assoc$, theo quy ước phần tử đầu tiên của
$associates(as)$ luôn là Group sở hữu.
\end{ghichu}

\[
\forall as\in Assoc:
\]
\[
\left(
\begin{array}{c}
associates(as)=\langle c_1,\ldots,c_n\rangle
\\[1mm]
\land\
\exists i,j\ (i\neq j\land c_i,c_j\in OrgClass)
\end{array}
\right)
\Longleftrightarrow
\left(
\begin{array}{c}
n=2
\\
\land\ as\in Composition
\\
\land\ c_1\in Group
\\
\land\ c_2\in OrgClass
\\
\land\ c_1\neq c_2
\end{array}
\right).
\tag{WF-6}
\]

\begin{ghichu}
WF-6 buộc mọi association giữa hai lớp thuộc $OrgClass=Role\cup Group$ phải
là composition nhị phân có đầu thứ nhất thuộc $Group$. Vì vậy association
Role--Role không hợp lệ: vế trái đúng nhưng vế phải không thể đúng.
\end{ghichu}

% ============================================================
% Quan hệ Group--Group suy dẫn từ Composition
% ============================================================

Đặt
\[
Comp_G
=
\left\{
(g_1,g_2)\in Group\times Group
\;\middle|\;
\exists as\in Composition:\ associates(as)=\langle g_1,g_2\rangle
\right\}.
\]

% ============================================================
% Tính duy nhất của Group sở hữu qua Composition
% ============================================================

\[
\forall x\in OrgClass:
\quad
\left|
\left\{
g\in Group
\;\middle|\;
\exists as\in Composition:\ associates(as)=\langle g,x\rangle
\right\}
\right|
\leq 1.
\tag{WF-7}
\]

% ============================================================
% Composition giữa các Group không tạo chu trình
% ============================================================

\[
\forall g\in Group:
\quad
g\notin Comp_G^{+}(g),
\tag{WF-8}
\]
trong đó $R^{+}$ là bao đóng bắc cầu của quan hệ $R$.
